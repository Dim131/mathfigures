\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}

\usepackage{fp}
\pagestyle{empty}

\definecolor{uuuuuu}{rgb}{0.26666666666666666,0.26666666666666666,0.26666666666666666}
\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}
\definecolor{ParacircleClr}{RGB}{217,185,7}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]

\def\Scale{2.5}
\def\La{1 * \Scale}
\def\Lb{2 * \Scale}
\def\Lc{\fpeval{sqrt(\La * \La + \Lb * \Lb)}}


\tkzDefPoints{0/0/B,1.1/3.7/A,5/0/C}

\tkzDefSpcTriangle[ex](A,B,C){Ja,Jb,Jc}
\tkzDefSpcTriangle[extouch](A,B,C){Ta,Tb,Tc}
\tkzDefTriangleCenter[nagel](A,B,C)

\tkzDefPointBy[projection=onto A--B](Ja)\tkzGetPoint{IA''}
\tkzDefPointBy[projection=onto A--C](Ja)\tkzGetPoint{IA'''}
\tkzDefPointBy[projection=onto B--A](Jb)\tkzGetPoint{IB''}
\tkzDefPointBy[projection=onto B--C](Jb)\tkzGetPoint{IB'''}
\tkzDefPointBy[projection=onto C--A](Jc)\tkzGetPoint{IC''}
\tkzDefPointBy[projection=onto C--B](Jc)\tkzGetPoint{IC'''}

\tkzFillPolygon[fill=white](A,B,C)
\tkzDrawLines[add=0 and 0](Ja,Jb Jb,Jc Jc,Ja)
\tkzDrawPoints[new](Ja,Jb,Jc,Ta,Tb,Tc)
\tkzDrawLines[add=1 and 1.75,dashed](A,B)
\tkzDrawLines[add=0.75 and 1,dashed](B,C)
\tkzDrawLines[add=1.75 and 1,dashed](C,A)

\tkzClipBB
\tkzDrawCircles[line width=0.5pt,color=ParacircleClr,fill=ParacircleClr,fill opacity=0.1](Ja,Ta Jb,Tb Jc,Tc)
\tkzDrawSegments[new,line width=0.5pt,color=black,dashed,dash pattern=on 1pt off 1.75pt](Ja,Ta Jb,Tb Jc,Tc Ja,IA'' Jb,IB'' Jc,IC'' Ja,IA''' Jb,IB''' Jc,IC''')

\tkzMarkRightAngles[line width=0.5pt, size=.15,color=AngleClr,fill=AngleClr,fill opacity=0.2](Ja,Ta,C Jb,Tb,A Jc,Tc,B Ja,IA''',C Ja,IA'',B Jb,IB'',A Jb,IB''',C Jc,IC'',A Jc,IC''',B)

\tkzDrawPolygon[color=ShapeClr](A,B,C)

\tkzDrawPoints[new,size=3](Ja,Jb,Jc)
\tkzDrawPoints[new,size=2](Ta,Tb,Tc)
\tkzDrawPoints[size=3](B,C,A)
\tkzDrawPoints[size=2](IA'',IA''',IB'',IB''',IC'',IC''')


\tkzLabelPoint[above](A){$\mathrm{A}$}
\tkzLabelPoint[below left](B){$\mathrm{B}$}
\tkzLabelPoint[below right](C){$\mathrm{\Gamma}$}

\tkzLabelPoint[below](Ta){$\mathrm{I_A'}$}
\tkzLabelPoint[above](Tb){$\mathrm{I_B'}$}
\tkzLabelPoint[above left](Tc){$\mathrm{I_\Gamma'}$}

\tkzLabelPoint[left](IA''){$\mathrm{I_A''}$}
\tkzLabelPoint[left](IB''){$\mathrm{I_B''}$}
\tkzLabelPoint[above right](IC''){$\mathrm{I_\Gamma''}$}

\tkzLabelPoint[above right](IA'''){$\mathrm{I_A'''}$}
\tkzLabelPoint[below](IB'''){$\mathrm{I_B'''}$}
\tkzLabelPoint[below](IC'''){$\mathrm{I_\Gamma'''}$}

\tkzLabelPoint[below](Ja){$\mathrm{J_A}$}
\tkzLabelPoint[above right](Jb){$\mathrm{J_B}$}
\tkzLabelPoint[above left](Jc){$\mathrm{J_\Gamma}$}


\end{tikzpicture}

\end{document}
