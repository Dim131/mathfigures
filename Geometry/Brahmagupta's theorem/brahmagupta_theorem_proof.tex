\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}
\pagestyle{empty}
\usepackage{fp}

\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}
\definecolor{BlueClr}{RGB}{71, 113, 158}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]
\tkzSetUpPoint[fill=black]

\tkzDefPoints{0/0/A,6/0/C,1.05/1.5/D}

\tkzDefCircle[circum](A,C,D)
\tkzGetPoint{O}
\tkzDrawCircle[line width=1.0pt, color=black](O,A)


% Find the fourth point of the quadrilateral.
\tkzDefLine[orthogonal=through D](C,A) \tkzGetPoint{X_1}
\tkzInterLC(D,X_1)(O,A) \tkzGetSecondPoint{B}

% Find the intersection of the diagonals.
\tkzInterLL(B,D)(A,C) \tkzGetPoint{I}

\tkzFillPolygon[fill=white](A,B,C,D)

\tkzMarkRightAngle[line width=0.5pt, size=.2,color=AngleClr,fill=AngleClr,fill opacity=0.2](B,I,A)

\tkzDefMidPoint(A,D) \tkzGetPoint{M}


\tkzInterLL(M,I)(B,C) \tkzGetPoint{H}

\tkzMarkRightAngle[line width=0.5pt, size=.2,color=AngleClr,fill=AngleClr,fill opacity=0.2](I,H,C)

% Mark angles.
\tkzFillAngles[mark=|,mksize=2,line width=1pt,size=.4,color=AngleClr,fill opacity=0.2](A,D,B A,C,B D,I,M)
\tkzMarkAngles[mark=|,mksize=2,line width=1pt,size=.4,color=AngleClr](A,D,B A,C,B D,I,M)
\tkzFillAngles[mark=||,mksize=2,line width=1pt,size=.4,color=BlueClr,fill opacity=0.2](C,A,D C,B,D M,I,A)
\tkzMarkAngles[mark=||,mksize=2,line width=1pt,size=.4,color=BlueClr](C,A,D C,B,D M,I,A)

\tkzDrawSegments[line width=0.5pt,color=black](A,C B,D H,M)


\tkzDrawPolygon[color=ShapeClr](A,B,C,D)
\tkzDrawPoints[size=3](A,B,C,D,I,M,H)
\tkzLabelPoint[left](A){$\rm A$}
\tkzLabelPoint[below](B){$\rm B$}
\tkzLabelPoint[right](C){$\rm \Gamma$}
\tkzLabelPoint[above](D){$\rm \Delta$}
\tkzLabelPoint[above right](I){$\rm I$}
\tkzLabelPoint[below right](H){$\rm H$}
\tkzLabelPoint[above left](M){$\rm M$}

\end{tikzpicture}

\end{document}
