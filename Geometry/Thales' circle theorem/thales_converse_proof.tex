\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}
\pagestyle{empty}

\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}
\definecolor{BlueClr}{RGB}{5,81,163}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]
\tkzSetUpPoint[fill=black]

\tkzDefPoints{0/0/B,1.8/2.4/A,5/0/C,2.5/0/O,2.5/0/M}

\tkzDefPointBy[projection=onto A--B](O)\tkzGetPoint{D}
\tkzDefPointBy[projection=onto A--C](O)\tkzGetPoint{E}

\tkzFillPolygon[fill=white](A,B,C)

\tkzFillAngles[fill=AngleClr,size=.4,fill opacity=0.2](M,B,A C,M,E)
\tkzMarkAngles[mark=||,mksize=2,line width=1pt,size=.4,color=AngleClr](M,B,A C,M,E)

\tkzFillAngles[fill=BlueClr,size=.4,fill opacity=0.2](D,M,B E,C,M)
\tkzMarkAngles[mark=|,mksize=1.5,line width=1pt,color=BlueClr,size=.4](D,M,B E,C,M)

\tkzMarkRightAngles[line width=1pt, size=.2,color=AngleClr,fill=AngleClr,fill opacity=0.2](B,A,C O,D,A O,E,A)

\tkzDrawSegments[line width=0.5pt,color=black,dashed,dash pattern=on 1pt off 1.75pt](O,B O,D O,E)

\tkzDrawPolygon[color=ShapeClr](A,B,C)
\tkzDrawPoints[size=3](A,B,C,D,E,O)
\tkzLabelPoint[above](A){$\rm A$}
\tkzLabelPoint[below left](B){$\rm B$}
\tkzLabelPoint[below right](C){$\rm \Gamma$}
\tkzLabelPoint[below](O){$\rm M$}
\tkzLabelPoint[above left](D){$\mathrm{\Delta}$}
\tkzLabelPoint[above right](E){$\mathrm{E}$}

\tkzMarkSegments[mark=s|,size=2](O,B O,C)
\tkzMarkSegments[mark=s||,size=2](O,E A,D A,D)


\end{tikzpicture}

\end{document}
