\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}
\pagestyle{empty}

\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]
\tkzSetUpPoint[fill=black]

\tkzDefPoints{0/0/A,6/0/B,1/2/D,3.5/2/C,2.5/0/E}

\tkzDefPointBy[projection=onto A--B](C)\tkzGetPoint{C'}

\tkzFillPolygon[fill=white](A,B,C,D)
\tkzMarkRightAngle[line width=1pt, size=.25,color=AngleClr,fill=AngleClr,fill opacity=0.2](C,C',B)

\tkzDrawPolygon[color=ShapeClr](A,B,C,D)
\tkzDrawPoints[size=3](A,B,C,D,E,B')
\tkzLabelPoint[below left](A){$\rm A$}
\tkzLabelPoint[below right](B){$\rm B$}
\tkzLabelPoint[above right](C){$\rm \Gamma$}
\tkzLabelPoint[above left](D){$\rm \Delta$}
\tkzLabelPoint[below](C'){$\rm \Gamma'$}
\tkzLabelPoint[below](E){$\rm E$}

\tkzLabelSegments[right](C,C'){$\upsilon$}

\tkzDrawSegments[line width=0.5pt,color=black,dashed,dash pattern=on 1pt off 1.75pt](C,E C,C')

\end{tikzpicture}
\end{document}
