\tkzDrawPoints[size=3](I,G,S,Na)
\tkzDrawPoints[size=3](B,C,A)

\tkzLabelPoint[above](A){$\rm A$}
\tkzLabelPoint[below left](B){$\rm B$}
\tkzLabelPoint[below ](C){$\rm \Gamma$}
\tkzLabelPoint[below,scale=0.7](G){$\rm G$}
