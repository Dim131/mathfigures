\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}

\tkzDrawPoints[size=3](A,B,C,D,E,Z,K,L)

\tkzLabelPoint[below left](K){$\rm K$}
\tkzLabelPoint[below](A){$\rm A$}

\tkzLabelPoint[below right](E){$\rm E$}
