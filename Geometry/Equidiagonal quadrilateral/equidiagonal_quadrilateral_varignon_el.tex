\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}
\pagestyle{empty}

\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]
\tkzSetUpPoint[fill=black]

\tkzDefPoint(180:2){A}
\tkzDefPoint(0:2){B}

\begin{scope}[shift=(A)]
\tkzDefPoint(25:6){C}
\end{scope}

\begin{scope}[shift=(B)]
\tkzDefPoint(145:6){D}
\end{scope}


\tkzFillPolygon[fill=white](A,B,C,D)

\tkzDefMidPoint(A,B) \tkzGetPoint{M1}
\tkzDefMidPoint(B,C) \tkzGetPoint{M2}
\tkzDefMidPoint(C,D) \tkzGetPoint{M3}
\tkzDefMidPoint(D,A) \tkzGetPoint{M4}

\tkzFillPolygon[fill=ShapeClr,fill opacity=0.1](M1,M2,M3,M4)

\tkzDrawSegments[line width=0.8pt,color=black](A,C B,D)


\tkzDrawPolygon[color=black](A,B,C,D)
\tkzDrawPolygon[color=ShapeClr](M1,M2,M3,M4)
\tkzDrawPoints[size=3](A,B,C,D)
\tkzDrawPoints[size=2](M1,M2,M3,M4)

\tkzLabelPoint[below left](A){$\rm A$}
\tkzLabelPoint[below right](B){$\rm B$}
\tkzLabelPoint[above right](C){$\rm \Gamma$}
\tkzLabelPoint[above left](D){$\rm \Delta$}

\tkzMarkSegments[mark=s||,size=2](M1,M2 M2,M3 M3,M4 M4,M1)

\end{tikzpicture}

\end{document}
