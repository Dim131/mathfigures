\documentclass[tikz]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.15}
\usepackage{mathrsfs}
\usetikzlibrary{arrows,calc}
\usepackage{tkz-euclide}
\pagestyle{empty}

\definecolor{AngleClr}{rgb}{0,0.39215686274509803,0}
\definecolor{ShapeClr}{rgb}{0.6,0.2,0}
\definecolor{BlueSqr}{RGB}{5,81,163}
\definecolor{RedSqr}{RGB}{0, 156, 0}

\begin{document}

\begin{tikzpicture}[scale=.75]
\tkzSetUpLine[line width=1pt,color=black]
\tkzSetUpPoint[fill=black]

\tkzDefPoints{0/0/B,3.5/0/C,0.8/1.8/A,2.75/2.25/D}

\tkzDefLine[bisector](B,A,D)\tkzGetPoint{a}
\tkzDefLine[bisector](C,B,A)\tkzGetPoint{b}
\tkzDefLine[bisector](D,C,B)\tkzGetPoint{c}
\tkzDefLine[bisector](A,D,C)\tkzGetPoint{d}

\tkzInterLL(A,a)(B,b) \tkzGetPoint{T1}
\tkzInterLL(B,b)(C,c) \tkzGetPoint{T2}
\tkzInterLL(C,c)(D,d) \tkzGetPoint{T3}
\tkzInterLL(D,d)(A,a) \tkzGetPoint{T4}

\tkzDefTriangleCenter[circum](T1,T2,T3) \tkzGetPoint{O}

\tkzFillPolygon[fill=white](A,B,C,D)
\tkzFillPolygon[fill=ShapeClr,fill opacity=0.1](T1,T2,T3,T4)
\tkzDrawCircle[line width=0.75,color=black](O,T1)

\tkzDrawSegments[line width=0.75pt,color=red](A,T4 B,T2 C,T2 D,T4)

\tkzDrawPolygon[color=ShapeClr](A,B,C,D)
\tkzDrawPoints[size=3](A,B,C,D)
\tkzDrawPoints[size=1](T1,T2,T3,T4)
\tkzLabelPoint[above left](A){$\mathrm{A}$}
\tkzLabelPoint[below left](B){$\rm B$}
\tkzLabelPoint[below right](C){$\rm \Gamma$}
\tkzLabelPoint[above right](D){$\rm \Delta$}

\tkzFillAngles[fill=AngleClr,size=.3,fill opacity=0.2](B,A,T2 T2,A,D)
\tkzMarkAngles[line width=1pt,size=.3,color=AngleClr,mark=|,mksize=1](B,A,T2 T2,A,D)

\tkzFillAngles[fill=AngleClr,size=.3,fill opacity=0.2](A,D,T4 T4,D,C)
\tkzMarkAngles[line width=1pt,size=.3,color=AngleClr,mark=||,mksize=1](A,D,T4 T4,D,C)

\tkzFillAngles[fill=AngleClr,size=.3,fill opacity=0.2](D,C,T2 T2,C,B)
\tkzMarkAngles[line width=1pt,size=.3,color=AngleClr,mark=s,mksize=1](D,C,T2 T2,C,B)

\tkzFillAngles[fill=AngleClr,size=.3,fill opacity=0.2](C,B,T2 T2,B,A)
\tkzMarkAngles[line width=1pt,size=.3,color=AngleClr,mark=s,mksize=1](C,B,T2 T2,B,A)

\end{tikzpicture}

\end{document}
